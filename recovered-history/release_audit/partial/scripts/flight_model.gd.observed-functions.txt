func forward() -> Vector3:
	return -(Basis(attitude).z)

func brake() -> void:
	target_speed = 0.0
	assisted = true
	autopilot = false

func can_dock() -> bool:
	return not surface_mode and not combat_active and hull>0 and station_available and position.distance_to(STATION) <= DOCK_RADIUS and velocity.length() <= DOCK_SPEED
